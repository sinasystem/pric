import unittest
from decimal import Decimal

from backtest.account import Account


class AccountTest(unittest.TestCase):

    def test_starting_balance(self):
        account = Account("10000.00")

        self.assertEqual(
            account.balance,
            Decimal("10000.00")
        )

        self.assertEqual(
            account.equity,
            Decimal("10000.00")
        )

    def test_realized_pnl_changes_balance(self):
        account = Account("10000.00")

        account.apply_pnl(Decimal("-25.50"))

        self.assertEqual(
            account.balance,
            Decimal("9974.50")
        )

        self.assertEqual(
            account.equity,
            Decimal("9974.50")
        )

    def test_floating_pnl_changes_equity_not_balance(self):
        account = Account("10000.00")

        account.update_floating_pnl(
            Decimal("-80.00")
        )

        self.assertEqual(
            account.balance,
            Decimal("10000.00")
        )

        self.assertEqual(
            account.floating_pnl,
            Decimal("-80.00")
        )

        self.assertEqual(
            account.equity,
            Decimal("9920.00")
        )

    def test_realized_and_floating_pnl_together(self):
        account = Account("10000.00")

        account.apply_pnl(Decimal("-30.00"))
        account.update_floating_pnl(
            Decimal("-80.00")
        )

        self.assertEqual(
            account.balance,
            Decimal("9970.00")
        )

        self.assertEqual(
            account.equity,
            Decimal("9890.00")
        )

import unittest
from datetime import datetime
from decimal import Decimal

from backtest.account import Account

class OneShotBuyStrategy:
    """
    Generates exactly one BUY signal after the first candle closes.
    """

    def __init__(self):
        self.called = 0

    def on_candle(self, candle):
        self.called += 1

        if self.called == 1:
            return Signal.BUY

        return Signal.NONE

class OneShotSellStrategy:
    """
    Generates exactly one SELL signal after the first candle closes.
    """

    def __init__(self):
        self.called = 0

    def on_candle(self, candle):
        self.called += 1

        if self.called == 1:
            return Signal.SELL

        return Signal.NONE

class MultiSignalStrategy:
    def __init__(self, signals):
        self.signals = signals
        self.index = 0

    def on_candle(self, candle):
        signal = self.signals[self.index]
        self.index += 1
        return signal

class ExecutionTimingTest(unittest.TestCase):

    def test_signal_executes_on_next_candle_open(self):
        """
        Verify:

        1. Signal is generated from candle N.
        2. Entry happens on candle N+1.
        3. BUY entry uses ASK.
        4. BUY exit uses BID.
        5. SL is evaluated using BID.
        6. Commission is deducted.
        7. P&L is calculated using the XAUUSD contract size.
        """

        candles = [
            Candle(
                datetime(2026, 1, 1, 10, 0),
                100.0,
                101.0,
                99.0,
                100.5,
            ),
            Candle(
                datetime(2026, 1, 1, 10, 5),
                105.0,
                105.5,
                104.5,
                105.2,
            ),
            Candle(
                datetime(2026, 1, 1, 10, 10),
                105.0,
                105.5,
                103.5,
                104.0,
            ),
        ]

        strategy = OneShotBuyStrategy()

        engine = BacktestEngine(
            candles=candles,
            instrument=XAUUSD,
            trading_config=TRADING_CONFIG,
            stop_loss_distance=1000,
            take_profit_distance=1000,
        )

        result = engine.run(strategy)

        self.assertEqual(len(result.completed_trades), 1)

        trade = result.completed_trades[0]

        # Signal timing
        self.assertEqual(
            trade.signal_time,
            datetime(2026, 1, 1, 10, 0),
        )

        # Entry happens on the next candle.
        self.assertEqual(
            trade.entry_time,
            datetime(2026, 1, 1, 10, 5),
        )

        # Candle open = 105.000
        # Spread = 0.008
        # Half spread = 0.004
        # BUY executes at ASK = 105.004
        self.assertEqual(
            trade.entry_ticks,
            105004,
        )

        # SL = 105.004 - 1.000 = 104.004
        self.assertEqual(
            trade.stop_loss_ticks,
            104004,
        )

        # TP = 105.004 + 1.000 = 106.004
        self.assertEqual(
            trade.take_profit_ticks,
            106004,
        )

        # Candle 3 reference low = 103.500
        #
        # BID low = 103.500 - 0.004
        #         = 103.496
        #
        # Therefore the BUY SL is hit.
        self.assertEqual(
            trade.exit_time,
            datetime(2026, 1, 1, 10, 10),
        )

        self.assertEqual(
            trade.exit_ticks,
            104004,
        )

        self.assertEqual(
            trade.exit_reason,
            "stop_loss",
        )

        # Entry: 105.004
        # Exit:  104.004
        #
        # Price movement = -1.000
        # = -1000 ticks
        self.assertEqual(
            trade.profit_ticks,
            -1000,
        )

        # 0.01 lot is used by the current configuration.
        self.assertEqual(
            trade.lot_size,
            Decimal("0.01"),
        )

        # 0.01 lot = 1 oz.
        #
        # $1.00 price movement × 1 oz = $1 loss.
        self.assertEqual(
            trade.gross_profit,
            Decimal("-1.000"),
        )

        # Commission:
        # $8 × 0.01 = $0.08
        self.assertEqual(
            trade.commission,
            Decimal("0.08"),
        )

        # Net:
        # -$1.00 - $0.08 = -$1.08
        self.assertEqual(
            trade.net_profit,
            Decimal("-1.080"),
        )

        # Starting balance = $10,000
        self.assertEqual(
            engine.account.balance,
            Decimal("9998.920"),
        )

        self.assertGreater(
            trade.entry_time,
            trade.signal_time,
        )

                2000.0,

        self.assertEqual(

        )

    def test_daily_loss_limit_blocks_new_trade(self):
        candles = [
            # 1. Signal BUY
            Candle(
                datetime(2026, 1, 1, 10, 0),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # 2. BUY executes here and immediately hits SL.
            Candle(
                datetime(2026, 1, 1, 10, 5),
                2000.0, 2000.5, 1899.0, 1999.0
            ),

            # 3. Signal BUY for a second trade.
            Candle(
                datetime(2026, 1, 1, 10, 10),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # 4. Second trade would execute here.
            Candle(
                datetime(2026, 1, 1, 10, 15),
                2000.0, 2000.5, 1999.5, 2000.0
            ),
        ]

        strategy = MultiSignalStrategy([
            Signal.BUY,
            Signal.NONE,
            Signal.BUY,
            Signal.NONE,
        ])

        engine = BacktestEngine(
            candles=candles,
            instrument=XAUUSD,
            trading_config=TRADING_CONFIG,
            stop_loss_distance=100000,
            take_profit_distance=10000,
        )

        # Make the first loss large enough to hit the $100 daily limit.
        engine.current_trade = None

        result = engine.run(strategy)

        self.assertEqual(len(result.completed_trades), 1)

        first_trade = result.completed_trades[0]

        self.assertEqual(first_trade.direction, "BUY")
        self.assertEqual(first_trade.exit_reason, "stop_loss")

        self.assertLessEqual(
            first_trade.net_profit,
            Decimal("-100.00"),
        )

        # The second trade must NOT have been opened.
        self.assertEqual(engine.open_trades_at_end, [])

        # The first trade's loss should remain realized.
        self.assertEqual(
            engine.account.floating_pnl,
            Decimal("0"),
        )
    def test_daily_loss_below_limit_allows_new_trade(self):
        candles = [
            # Signal trade #1.
            Candle(
                datetime(2026, 1, 1, 10, 0),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # Trade #1 executes and hits its SL.
            Candle(
                datetime(2026, 1, 1, 10, 5),
                2000.0, 2000.5, 1949.0, 1999.0
            ),

            # Signal trade #2.
            Candle(
                datetime(2026, 1, 1, 10, 10),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # Trade #2 executes here and remains open.
            Candle(
                datetime(2026, 1, 1, 10, 15),
                2000.0, 2000.5, 1999.5, 2000.0
            ),
        ]

        strategy = MultiSignalStrategy([
            Signal.BUY,
            Signal.NONE,
            Signal.BUY,
            Signal.NONE,
        ])

        engine = BacktestEngine(
            candles=candles,
            instrument=XAUUSD,
            trading_config=TRADING_CONFIG,
            stop_loss_distance=50000,
            take_profit_distance=100000,
        )

        result = engine.run(strategy)

        # Trade #1 hit its SL.
        self.assertEqual(len(result.completed_trades), 1)

        first_trade = result.completed_trades[0]

        self.assertEqual(first_trade.direction, "BUY")
        self.assertEqual(first_trade.exit_reason, "stop_loss")

        # Gross loss is approximately $50, plus $0.08 commission.
        self.assertEqual(
            first_trade.net_profit,
            Decimal("-50.080"),
        )

        # Trade #2 was allowed and is still open at the end.
        self.assertGreater(len(result.open_trades), 0)
        self.assertEqual(
            result.open_trades[0].direction,
            "BUY",
        )

        # Its entry should be on the fourth candle.
        self.assertEqual(
            result.open_trades[0].entry_time,
            datetime(2026, 1, 1, 10, 15),
        )

    def test_daily_loss_limit_includes_realized_and_floating_pnl(self):
        candles = [
            # Signal trade #1.
            Candle(
                datetime(2026, 1, 1, 10, 0),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # Trade #1 executes and hits SL for roughly $60 loss.
            Candle(
                datetime(2026, 1, 1, 10, 5),
                2000.0, 2000.5, 1939.0, 1999.0
            ),

            # Signal trade #2.
            Candle(
                datetime(2026, 1, 1, 10, 10),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # Trade #2 executes.
            Candle(
                datetime(2026, 1, 1, 10, 15),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # Trade #2 moves against us by about $41.
            Candle(
                datetime(2026, 1, 1, 10, 20),
                1959.0, 1960.0, 1958.0, 1959.0
            ),

            # Signal trade #3.
            Candle(
                datetime(2026, 1, 1, 10, 25),
                2000.0, 2000.5, 1999.5, 2000.0
            ),

            # Trade #3 would execute here if risk allowed it.
            Candle(
                datetime(2026, 1, 1, 10, 30),
                2000.0, 2000.5, 1999.5, 2000.0
            ),
        ]

        strategy = MultiSignalStrategy([
            Signal.BUY,
            Signal.NONE,
            Signal.BUY,
            Signal.NONE,
            Signal.NONE,
            Signal.BUY,
            Signal.NONE,
        ])

        engine = BacktestEngine(
            candles=candles,
            instrument=XAUUSD,
            trading_config=TRADING_CONFIG,
            stop_loss_distance=60000,
            take_profit_distance=100000,
        )

        result = engine.run(strategy)

        # Trade #1 should have closed at its SL.
        self.assertEqual(len(result.completed_trades), 1)

        first_trade = result.completed_trades[0]

        self.assertEqual(first_trade.exit_reason, "stop_loss")

        # Trade #1 loses $60 + $0.08 commission.
        self.assertEqual(
            first_trade.net_profit,
            Decimal("-60.080"),
        )

        # Trade #2 was allowed because realized loss was still
        # below the $100 daily limit.
        self.assertGreater(len(engine.open_trades_at_end), 0)

        # Trade #3 must NOT have been opened because:
        #
        # realized ≈ -$60.08
        # floating ≈ -$41
        # total < -$100
        #
        # The existing trade (#2) should remain open.
        self.assertEqual(
            engine.open_trades_at_end[0].entry_time,
            datetime(2026, 1, 1, 10, 15),
        )

    def test_weekly_loss_limit_resets_on_new_week(self):
        risk_manager = RiskManager(TRADING_CONFIG)

        # Monday, January 5, 2026.
        monday = datetime(2026, 1, 5, 10, 0)

        # Create a completed trade with a $400 loss.
        trade = Trade(
            direction="BUY",
            lot_size=Decimal("0.01"),
            signal_time=monday,
            entry_time=monday,
            entry_ticks=2000000,
            stop_loss_ticks=1900000,
            take_profit_ticks=2100000,
            exit_time=monday,
            exit_ticks=1900000,
            gross_profit=Decimal("-400.00"),
            commission=Decimal("0.00"),
            net_profit=Decimal("-400.00"),
            profit_ticks=-100000,
            exit_reason="stop_loss",
        )

        risk_manager.update(trade)

        # The weekly limit has been reached.
        self.assertTrue(
            risk_manager.weekly_loss_reached()
        )

        # Move to the following Monday.
        next_monday = datetime(2026, 1, 12, 10, 0)

        # A new week should reset the weekly realized P&L.
        self.assertTrue(
            risk_manager.can_open_trade(next_monday)
        )

        self.assertEqual(
            risk_manager.weekly_realized_pnl,
            Decimal("0"),
        )

    def test_loss_limit_does_not_force_close_existing_position(self):
        """
        Verify that reaching the daily loss limit prevents new trades
        but does not forcibly close an already-open position.
        """

        class BuyAndHoldStrategy:
            def __init__(self):
                self.called = 0

            def on_candle(self, candle):
                self.called += 1

                if self.called == 1:
                    return Signal.BUY

                return Signal.NONE

        candles = [
            Candle(
                datetime(2026, 1, 5, 10, 0),
                2000.0,
                2000.5,
                1999.5,
                2000.0,
            ),
            Candle(

class TestKeyBarDetector(unittest.TestCase):
    def setUp(self):
        self.start = datetime(2026, 1, 1, 10, 0)

    def make_candle(
        self,
        offset: int,
        open_price: str,
        high: str,
        low: str,
        close: str,
    ) -> Candle:
        return Candle(
            timestamp=self.start + timedelta(minutes=5 * offset),
            open=float(open_price),
            high=float(high),
            low=float(low),
            close=float(close),
        )

    def make_signal_bar(
        self,
        direction: str,
        signal_offset: int = 10,
        signal_open: str = "104",
        signal_high: str = "108",
        signal_low: str = "101",
        signal_close: str = "106",
    ) -> SignalBarEvent:

        pullback = self.make_candle(
            3,
            "105",
            "110",
            "100",
            "102",
        )

        breakout = self.make_candle(
            1,
            "100",
            "108",
            "100",
            "107",
        )

        pre_breakout = self.make_candle(
            0,
            "99",
            "100",
            "97",
            "99",
        )

        follow_through = self.make_candle(
            2,
            "107",
            "110",
            "105",
            "109",
        )

        p_gap = PGapEvent(
            direction=direction,
            pre_breakout_candle=pre_breakout,
            breakout_candle=breakout,
            follow_through_candle=follow_through,
        )

        second_leg = SecondLegEvent(
            direction=direction,
            p_gap=p_gap,
            pullback_candle=pullback,
            reference_candle=pre_breakout,
        )

        structure_candle = self.make_candle(
            9,
            "105",
            "105",
            "105",
            "105",
        )

        structure_point = StructurePoint(
            type=(
                "lower_high"
                if direction == "buy"
                else "higher_low"
            ),
            candle=structure_candle,
            price=105,
        )

        pullback_structure = PullbackStructureEvent(
            direction=direction,
            second_leg=second_leg,
            anchor_candle=pullback,
            structure_points=[structure_point],
        )

        signal_candle = self.make_candle(
            signal_offset,
            signal_open,
            signal_high,
            signal_low,
            signal_close,
        )

        return SignalBarEvent(
            direction=direction,
            pullback_structure=pullback_structure,
            signal_candle=signal_candle,
        )

import unittest
from datetime import datetime, timedelta
from decimal import Decimal

from models.candle import Candle
from strategies.p_gap import PGapDetector, PGapEvent
from strategies.spike_event import SpikeEvent
from strategies.spike_measurements import SpikeMeasurements


class TestPGapDetector(unittest.TestCase):
    def setUp(self):
        self.detector = PGapDetector()
        self.start = datetime(2026, 1, 1, 10, 0)

    def make_candle(
        self,
        offset: int,
        open_price: str,
        high: str,
        low: str,
        close: str,
    ) -> Candle:
        return Candle(
            timestamp=self.start + timedelta(minutes=5 * offset),
            open=float(open_price),
            high=float(high),
            low=float(low),
            close=float(close),
        )

    def make_spike_event(
        self,
        breakout_candle: Candle,
        direction: str,
    ) -> SpikeEvent:
        zero = Decimal("0")

        measurements = SpikeMeasurements(
            direction=direction,
            total_range=zero,
            body=zero,
            upper_wick=zero,
            lower_wick=zero,
            body_ratio_of_range=zero,
            closing_wick_ratio=zero,
            average_previous_range=zero,
            average_previous_body=zero,
            range_ratio=zero,
            body_ratio=zero,
            previous_high=zero,
            previous_low=zero,
            high_breakout_distance=zero,
            low_breakout_distance=zero,
            close_breakout_distance=zero,
            breakout_distance=zero,
            is_spike=True,
        )

        return SpikeEvent(
            breakout_candle=breakout_candle,
            direction=direction,
            measurements=measurements,
        )

    def test_detects_valid_bullish_p_gap(self):
        # Pre-breakout candle high = 100.0
        # Spike breaks upward.
        # Follow-through low = 101.0
        # 101.0 > 100.0 -> valid bullish P-Gap.

        pre_breakout = self.make_candle(
            0, "99.0", "100.0", "98.0", "99.5"
        )

        breakout = self.make_candle(
            1, "99.5", "105.0", "99.5", "104.0"
        )

        follow_through = self.make_candle(
            2, "104.0", "108.0", "101.0", "107.0"
        )

        spike_event = self.make_spike_event(
            breakout,
            "buy",
        )

        result = self.detector.detect(
            spike_event=spike_event,
            pre_breakout_candle=pre_breakout,
            follow_through_candle=follow_through,
        )

        self.assertIsNotNone(result)
        self.assertIsInstance(result, PGapEvent)
        self.assertEqual(result.direction, "buy")
        self.assertEqual(result.pre_breakout_candle, pre_breakout)
        self.assertEqual(result.breakout_candle, breakout)
        self.assertEqual(result.follow_through_candle, follow_through)

    def test_rejects_bullish_shadow_overlap(self):
        # Pre-breakout high = 100.0
        # Follow-through low = 100.0
        # Exact contact means the shadows overlap -> invalid P-Gap.

        pre_breakout = self.make_candle(
            0, "99.0", "100.0", "98.0", "99.5"
        )

        breakout = self.make_candle(

        )

        follow_through = self.make_candle(
            2, "104.0", "108.0", "100.0", "107.0"
        )

        spike_event = self.make_spike_event(
            breakout,
            "buy",
        )

        result = self.detector.detect(
            spike_event=spike_event,
            pre_breakout_candle=pre_breakout,
            follow_through_candle=follow_through,
        )

        self.assertIsNone(result)

    def test_detects_valid_bearish_p_gap(self):
        # Pre-breakout candle low = 98.0
        # Spike breaks downward.
        # Follow-through high = 97.0

            p_gap=p_gap,

            type=structure_type,
            candle=structure_candle,
            price=Decimal(latest_structure_price),

            direction=direction,
            second_leg=second_leg,
            anchor_candle=pullback,
            structure_points=[structure_point],

    return Candle(
        timestamp=BASE_TIME + timedelta(minutes=minute),
        open=open_price,
        high=high,
        low=low,
        close=close,
    )


def make_spike(candle: Candle) -> SpikeEvent:
    measurements = SpikeMeasurements(
        direction="buy",
        total_range=2.0,
        body=1.5,
        upper_wick=0.2,
        lower_wick=0.3,
        body_ratio_of_range=0.75,
        closing_wick_ratio=0.10,
        average_previous_range=1.0,
        average_previous_body=0.8,
        range_ratio=2.0,
        body_ratio=1.875,
        previous_high=99.0,
        previous_low=98.0,
        high_breakout_distance=2.0,
        low_breakout_distance=0.0,
        close_breakout_distance=1.5,
        is_spike=True,
        breakout_distance=1.5,
    )

    return SpikeEvent(
        breakout_candle=candle,
        direction="buy",
        measurements=measurements,
    )


def make_p_gap(
    spike: SpikeEvent,
    pre_breakout: Candle,
    follow_through: Candle,
) -> PGapEvent:
    return PGapEvent(
        direction="buy",
        pre_breakout_candle=pre_breakout,
        breakout_candle=spike.breakout_candle,
        follow_through_candle=follow_through,
    )


def make_second_leg(
    p_gap: PGapEvent,
    reference: Candle,
    anchor: Candle,
) -> SecondLegEvent:
    return SecondLegEvent(
        direction="buy",
        p_gap=p_gap,
        pullback_candle=anchor,
        reference_candle=reference,
    )


def make_pullback_structure(
    second_leg: SecondLegEvent,
    anchor: Candle,
) -> PullbackStructureEvent:
    structure_points = [

import unittest
from datetime import datetime, timedelta
from decimal import Decimal

from models.candle import Candle
from strategies.swing_processor import SwingProcessor


class TestSwingProcessor(unittest.TestCase):
    def setUp(self):
        self.processor = SwingProcessor()
        self.start = datetime(2026, 1, 1, 10, 0)

    def candle(
        self,
        offset: int,
        high: str,
        low: str,
    ) -> Candle:
        return Candle(
            timestamp=self.start + timedelta(minutes=5 * offset),
            open=float(low),
            high=float(high),
            low=float(low),
            close=float(low),
        )

    def test_waits_for_three_candles(self):
        result = self.processor.process(
            self.candle(0, "100", "98")
        )

        self.assertIsNone(result)

        result = self.processor.process(
            self.candle(1, "105", "99")
        )

        self.assertIsNone(result)

        self.assertEqual(
            self.processor.history.points,
            [],
        )

    def test_confirms_middle_candle_as_swing_high(self):
        self.processor.process(
            self.candle(0, "100", "98")
        )
        self.processor.process(
            self.candle(1, "105", "99")
        )

        result = self.processor.process(
            self.candle(2, "102", "100")
        )

        self.assertIsNotNone(result)
        self.assertEqual(result.type, "high")
        self.assertEqual(
            result.candle.timestamp,
            self.start + timedelta(minutes=5),
        )
        self.assertEqual(
            result.price,
            Decimal("105"),
        )

        self.assertEqual(
            self.processor.history.points,
            [result],
        )

    def test_confirms_middle_candle_as_swing_low(self):
        self.processor.process(
            self.candle(0, "105", "100")
        )
        self.processor.process(
            self.candle(1, "103", "95")
        )

        result = self.processor.process(
            self.candle(2, "102", "98")
        )

        self.assertIsNotNone(result)
        self.assertEqual(result.type, "low")
        self.assertEqual(
            result.candle.timestamp,
            self.start + timedelta(minutes=5),
        )
        self.assertEqual(
            result.price,
            Decimal("95"),
        )

        self.assertEqual(
            self.processor.history.points,
            [result],
        )

    def test_non_swing_does_not_modify_history(self):
        candles = [
            self.candle(0, "100", "98"),
            self.candle(1, "103", "99"),
            self.candle(2, "105", "100"),
        ]

        for candle in candles:
            result = self.processor.process(candle)

        self.assertIsNone(result)
        self.assertEqual(
            self.processor.history.points,
            [],
        )

    def test_processes_multiple_confirmed_swings(self):
        candles = [
            self.candle(0, "100", "98"),
            self.candle(1, "105", "99"),
            self.candle(2, "102", "100"),
            self.candle(3, "101", "97"),
            self.candle(4, "103", "99"),
        ]

        results = []

        for candle in candles:
            result = self.processor.process(candle)

            if result is not None:
                results.append(result)

        self.assertEqual(len(results), 2)

        self.assertEqual(
            results[0].type,
            "high",
        )
        self.assertEqual(
            results[0].price,
            Decimal("105"),
        )

        self.assertEqual(
            results[1].type,
            "low",
        )
        self.assertEqual(
            results[1].price,
            Decimal("97"),
        )

        self.assertEqual(
            self.processor.history.points,
            results,
        )

    def test_confirmed_swing_is_one_candle_behind_current_candle(self):
        self.processor.process(
            self.candle(0, "100", "98")
        )
        self.processor.process(
            self.candle(1, "105", "99")
        )

        result = self.processor.process(
            self.candle(2, "102", "100")
        )

        self.assertIsNotNone(result)

        self.assertEqual(
            result.candle.timestamp,
            self.start + timedelta(minutes=5),
        )

        self.assertNotEqual(
            result.candle.timestamp,
            self.start + timedelta(minutes=10),
        )

import unittest
from datetime import datetime, timedelta

from models.candle import Candle
from models.timeframe import M15Builder


def make_candle(
    hour: int,
    minute: int,
    open_: float,
    high: float,
    low: float,
    close: float,
) -> Candle:
    return Candle(
        timestamp=datetime(2026, 1, 1, hour, minute),
        open=open_,
        high=high,
        low=low,
        close=close,
    )


class TestM15Builder(unittest.TestCase):

    def test_aligned_candles_create_m15_candle(self):
        builder = M15Builder()

        self.assertIsNone(
            builder.update(
                make_candle(9, 0, 100, 105, 99, 103)
            )
        )

        self.assertIsNone(
            builder.update(
                make_candle(9, 5, 103, 108, 102, 107)
            )
        )

        result = builder.update(
            make_candle(9, 10, 107, 110, 104, 109)
        )

        self.assertIsNotNone(result)

    def test_misaligned_candles_are_ignored_until_next_boundary(self):
        builder = M15Builder()

        self.assertIsNone(
            builder.update(
                make_candle(9, 5, 100, 105, 99, 103)
            )
        )

        self.assertIsNone(
            builder.update(
                make_candle(9, 10, 103, 108, 102, 107)
            )
        )

        # 09:15 is the first valid M15 starting point.
        self.assertIsNone(
            builder.update(
                make_candle(9, 15, 107, 110, 104, 109)
            )
        )

        self.assertIsNone(
            builder.update(
                make_candle(9, 20, 109, 112, 108, 111)
            )
        )

        result = builder.update(
            make_candle(9, 25, 111, 115, 110, 114)
        )

        self.assertIsNotNone(result)
        self.assertEqual(
            result.timestamp,
            datetime(2026, 1, 1, 9, 15),
        )

    def test_m15_ohlc_is_aggregated_correctly(self):
        builder = M15Builder()

        builder.update(
            make_candle(9, 0, 100, 105, 99, 103)
        )
        builder.update(
            make_candle(9, 5, 103, 108, 102, 107)
        )

        result = builder.update(
            make_candle(9, 10, 107, 110, 104, 109)
        )

        self.assertEqual(
            result.timestamp,
            datetime(2026, 1, 1, 9, 0),
        )
        self.assertEqual(result.open, 100)
        self.assertEqual(result.high, 110)
        self.assertEqual(result.low, 99)
        self.assertEqual(result.close, 109)

    def test_gap_invalidates_partial_group(self):
        builder = M15Builder()

        # Start valid group.
        builder.update(
            make_candle(9, 15, 100, 105, 99, 103)
        )

        builder.update(
            make_candle(9, 20, 103, 108, 102, 107)
        )

        # 09:25 is missing. 09:30 arrives instead.
        self.assertIsNone(
            builder.update(
                make_candle(9, 30, 107, 110, 104, 109)
            )
        )

        # The previous partial group must not be completed.
        self.assertEqual(len(builder.candles), 0)

    def test_after_gap_next_aligned_candle_can_start_new_group(self):
        builder = M15Builder()

        builder.update(
            make_candle(9, 15, 100, 105, 99, 103)
        )

        builder.update(
            make_candle(9, 20, 103, 108, 102, 107)
        )

        # Gap: 09:25 missing.
        builder.update(
            make_candle(9, 30, 107, 110, 104, 109)
        )

        # 09:30 starts a fresh group.
        builder.update(
            make_candle(9, 35, 109, 112, 108, 111)
        )

        result = builder.update(
            make_candle(9, 40, 111, 115, 110, 114)
        )

        self.assertIsNotNone(result)
        self.assertEqual(
            result.timestamp,
            datetime(2026, 1, 1, 9, 30),
        )

    def test_invalid_spacing_does_not_create_m15_candle(self):
        builder = M15Builder()

        builder.update(
            make_candle(9, 0, 100, 105, 99, 103)
        )

        # 09:10 is not the expected second candle.
        self.assertIsNone(
            builder.update(
                make_candle(9, 10, 103, 108, 102, 107)
            )
        )

        self.assertEqual(len(builder.candles), 0)

    def test_hour_boundary_is_valid(self):
        builder = M15Builder()

        builder.update(
            make_candle(9, 45, 100, 105, 99, 103)
        )

        builder.update(
            make_candle(9, 50, 103, 108, 102, 107)
        )

        result = builder.update(
            make_candle(9, 55, 107, 110, 104, 109)
        )

        self.assertIsNotNone(result)

        self.assertEqual(
            result.timestamp,
            datetime(2026, 1, 1, 9, 45),
        )

    def test_m15_history_is_limited_to_max_history(self):
        builder = M15Builder(max_history=2)

        start = datetime(2026, 1, 1, 9, 0)

        for block in range(3):
            for offset in range(3):
                timestamp = start + timedelta(
                    minutes=15 * block + 5 * offset
                )

                builder.update(
                    Candle(
                        timestamp=timestamp,
                        open=100 + block,
                        high=105 + block,
                        low=95 + block,
                        close=102 + block,
                    )
                )

        self.assertEqual(len(builder.candles), 2)

        self.assertEqual(
            builder.candles[0].timestamp,
            datetime(2026, 1, 1, 9, 15),
        )

        self.assertEqual(
            builder.candles[1].timestamp,
            datetime(2026, 1, 1, 9, 30),
        )
