from config.datasets import DATASETS
from config.instrument import XAUUSD
from config.trading import TRADING_CONFIG
from data.loader import load_dataset

from backtest.engine import BacktestEngine
from backtest.metrics import MetricsCalculator
from strategies.builders import build_sp2l_strategy
from analysis.sp2l_audit_csv import export_audit_events


def main():
    config = DATASETS["A"]

    candles = load_dataset(
        config.path,
        config.name,
    )

    strategy = build_sp2l_strategy()

    engine = BacktestEngine(
        candles=candles,
        instrument=XAUUSD,
        trading_config=TRADING_CONFIG,

        # 5.000 XAUUSD = 5,000 ticks
        stop_loss_distance=5000,

        # 10.000 XAUUSD = 10,000 ticks
        take_profit_distance=10000,
    )

    result = engine.run(strategy)
    
    successful_key_bar_times = {
        event.timestamp
        for event in strategy.audit_events
        if event.event == "KEY_BAR"
        and event.result == "SUCCESS"
    }

    executed_signal_times = {
        trade.signal_time
        for trade in result.completed_trades
    }

    # Iterate through the array of open trades
    for trade in result.open_trades:
        executed_signal_times.add(trade.signal_time)

    missing_from_trades = (
        successful_key_bar_times
        - executed_signal_times
    )

    print("\n--- Audit / Trade Comparison ---")
    print(f"Successful KEY_BAR setups: {len(successful_key_bar_times)}")
    print(f"Executed trade signals:    {len(executed_signal_times)}")
    print(f"Recognized but skipped:    {len(missing_from_trades)}")

    if missing_from_trades:
        print("\nSkipped Signals (Likely blocked by existing open trade or risk limits):")
        for timestamp in sorted(missing_from_trades):
            print(f" - {timestamp}")

    export_audit_events(
        strategy.audit_events,
        "analysis/output/sp2l_audit.csv",
    )
    
    metrics = MetricsCalculator().calculate(result)

    print("=" * 60)
    print("BACKTEST RESULTS")
    print("=" * 60)

    print(f"Dataset:           {config.name}")
    print(f"Candles:           {len(candles):,}")

    print("\n--- Account ---")
    print(f"Starting balance:  ${metrics.starting_balance:,.2f}")
    print(f"Ending balance:    ${metrics.ending_balance:,.2f}")
    print(f"Net profit:        ${metrics.net_profit:,.2f}")
    print(f"Return:            {metrics.return_percent:.2f}%")

    print("\n--- Trades ---")
    print(f"Total trades:      {metrics.total_trades:,}")
    print(f"Winning trades:    {metrics.winning_trades:,}")
    print(f"Losing trades:     {metrics.losing_trades:,}")
    print(f"Win rate:          {metrics.win_rate_percent:.2f}%")
    print(f"Profit factor:     {metrics.profit_factor:.2f}")
    print(f"Average trade:     ${metrics.average_trade:,.3f}")
    print(f"Largest win:       ${metrics.largest_win:,.3f}")
    print(f"Largest loss:      ${metrics.largest_loss:,.3f}")
    print(
        f"Max losing streak: "
        f"{metrics.max_consecutive_losses}"
    )

    print("\n--- Risk ---")
    print(f"Max drawdown:      ${metrics.max_drawdown:,.2f}")
    print(f"Max drawdown:      {metrics.max_drawdown_percent:.2f}%")
    print(f"Exposure:          {metrics.exposure_percent:.2f}%")
    print(
        f"Time in market:    "
        f"{metrics.total_time_in_market}"
    )

    print("\n--- Open Positions ---")

    # Handle the array of open trades
    if result.open_trades:
        print(f"Open trades:       {len(result.open_trades)}")
        for i, trade in enumerate(result.open_trades, 1):
            print(f"  {i}. {trade.direction} | Lot: {trade.lot_size} | Entry: {trade.entry_time}")
        print(
            f"Final floating P&L: "
            f"${result.final_floating_pnl:,.2f}"
        )
    else:
        print("Open trades:       0")

    print("=" * 60)


if __name__ == "__main__":
    main()

import os

def concatenate_py_files(output_filename="consolidated_code.txt"):
    parent_dir = os.path.abspath(os.path.join(os.getcwd(), "."))
    print(parent_dir)
    # Directories to completely ignore
    exclude_dirs = {'__pycache__', 'venv', '.venv', 'env', 'Lib', 'site-packages','tests'}
    
    with open(output_filename, 'w', encoding='utf-8') as outfile:
        for root, dirs, files in os.walk(parent_dir):
            # Modify dirs in-place to skip excluded directories
            dirs[:] = [d for d in dirs if d not in exclude_dirs]
                
            for file in files:
                if file.endswith(".py"):
                    file_path = os.path.join(root, file)
                    relative_path = os.path.relpath(file_path, parent_dir)
                    
                    try:
                        with open(file_path, 'r', encoding='utf-8') as infile:
                            content = infile.read()
                            
                            outfile.write(f"{'='*80}\n")
                            outfile.write(f"FILE: {relative_path}\n")
                            outfile.write(f"{'='*80}\n\n")
                            outfile.write(content)
                            outfile.write("\n\n")
                    except Exception as e:
                        print(f"Skipping {relative_path} due to error: {e}")
                        
    print(f"Done! Merged into '{output_filename}'.")

if __name__ == "__main__":
    concatenate_py_files()

from collections import Counter
from datetime import timedelta

from config.datasets import DATASETS
from data.loader import load_dataset


EXPECTED_INTERVAL = timedelta(minutes=5)


def analyze_dataset(dataset_key: str) -> None:
    config = DATASETS[dataset_key]
    candles = load_dataset(config.path, config.name)

    print("\n" + "=" * 70)
    print(f"DATASET: {config.name}")
    print("=" * 70)

    print(f"Candles: {len(candles):,}")
    print(f"Start:   {candles[0].timestamp}")
    print(f"End:     {candles[-1].timestamp}")

    analyze_intervals(candles)
    analyze_weekends(candles)
    analyze_suspicious_gaps(candles)
    analyze_duplicates(candles)
    analyze_price_data(candles)


def analyze_intervals(candles) -> None:
    intervals = []

    for previous, current in zip(candles, candles[1:]):
        intervals.append(current.timestamp - previous.timestamp)

    counts = Counter(intervals)

    print("\n--- Candle intervals ---")

    for interval, count in counts.most_common(15):
        print(f"{str(interval):>15} : {count:,}")

    normal = counts.get(EXPECTED_INTERVAL, 0)

    print(f"\nNormal 5-minute transitions: {normal:,}")
    print(f"Non-5-minute transitions:   {len(intervals) - normal:,}")

    print("\nLargest gaps:")

    large_gaps = sorted(
        (
            (interval, index)
            for index, interval in enumerate(intervals)
            if interval > EXPECTED_INTERVAL
        ),
        reverse=True,
    )[:20]

    for interval, index in large_gaps:
        previous = candles[index]
        current = candles[index + 1]

        print(
            f"{previous.timestamp} -> {current.timestamp} "
            f"= {interval}"
        )


def analyze_weekends(candles) -> None:
    print("\n--- Weekend gaps ---")

    weekend_gaps = []

    for previous, current in zip(candles, candles[1:]):
        interval = current.timestamp - previous.timestamp

        if interval <= EXPECTED_INTERVAL:
            continue

        # Python:
        # Monday = 0
        # Saturday = 5
        # Sunday = 6

        if (
            previous.timestamp.weekday() >= 4
            or current.timestamp.weekday() >= 5
        ):
            weekend_gaps.append(
                (
                    previous.timestamp,
                    current.timestamp,
                    interval,
                )
            )

    print(f"Weekend-related gaps: {len(weekend_gaps):,}")

    for previous, current, interval in weekend_gaps[:10]:
        print(
            f"{previous} -> {current} = {interval}"
        )


def analyze_duplicates(candles) -> None:
    print("\n--- Duplicate timestamps ---")

    timestamps = [candle.timestamp for candle in candles]
    counts = Counter(timestamps)

    duplicates = [
        (timestamp, count)
        for timestamp, count in counts.items()
        if count > 1
    ]

    print(f"Duplicate timestamps: {len(duplicates):,}")

    for timestamp, count in duplicates[:20]:
        print(f"{timestamp}: {count} occurrences")


def analyze_price_data(candles) -> None:
    print("\n--- Price data ---")

    invalid = []

    for candle in candles:
        if candle.high < max(candle.open, candle.close):
            invalid.append(candle)
            continue

        if candle.low > min(candle.open, candle.close):
            invalid.append(candle)
            continue

        if candle.low > candle.high:
            invalid.append(candle)

    print(f"Invalid OHLC candles: {len(invalid):,}")

    if invalid:
        for candle in invalid[:10]:
            print(candle)

def analyze_suspicious_gaps(candles) -> None:
    print("\n--- Suspicious gaps ---")

    suspicious = []

    for previous, current in zip(candles, candles[1:]):
        interval = current.timestamp - previous.timestamp

        if interval <= EXPECTED_INTERVAL:
            continue

        previous_weekday = previous.timestamp.weekday()
        current_weekday = current.timestamp.weekday()

        # Ignore obvious weekend closures.
        if previous_weekday == 4 and current_weekday in (6, 0):
            continue

        suspicious.append(
            (previous.timestamp, current.timestamp, interval)
        )

    print(f"Suspicious gaps: {len(suspicious):,}")

    for previous, current, interval in suspicious[:50]:
        print(
            f"{previous} -> {current} = {interval}"
        )

    analyze_dataset("B")

import csv
from datetime import datetime
from pathlib import Path
from collections import defaultdict

from config.datasets import DATASETS

class SetupInspector:
    def __init__(self, dataset_key: str, audit_path: str | Path):
        self.config = DATASETS[dataset_key]
        self.candles = load_dataset(self.config.path, self.config.name)
        self.audit_path = Path(audit_path)
        
        # Map timestamp to index for O(1) lookups
        self._candle_indices = {
            candle.timestamp: idx for idx, candle in enumerate(self.candles)
        }
        self.setups = self._load_audit_events()

    def _load_audit_events(self) -> dict[int, list[dict]]:
        setups = defaultdict(list)
        
        if not self.audit_path.exists():
            print(f"Audit file not found: {self.audit_path}")
            return dict()

        with self.audit_path.open("r", encoding="utf-8") as file:
            reader = csv.DictReader(file)
            for row in reader:
                setup_id_str = row.get("setup_id", "")
                if not setup_id_str:
                    continue
                    
                setup_id = int(setup_id_str)
                row["timestamp"] = datetime.fromisoformat(row["timestamp"])
                setups[setup_id].append(row)
                
        return dict(setups)

    def inspect(self, setup_id: int, pad_before: int = 10, pad_after: int = 5) -> None:
        if setup_id not in self.setups:
            print(f"Setup {setup_id} not found in audit log.")
            return

        events = self.setups[setup_id]
        
        # Determine the boundaries of the setup
        timestamps = [e["timestamp"] for e in events]
        start_time = min(timestamps)
        end_time = max(timestamps)

        start_idx = self._candle_indices.get(start_time)
        end_idx = self._candle_indices.get(end_time)

        if start_idx is None or end_idx is None:
            print(f"Setup {setup_id} timestamps do not align with dataset candles.")
            return

        # Apply padding
        slice_start = max(0, start_idx - pad_before)
        slice_end = min(len(self.candles) - 1, end_idx + pad_after)

        print("=" * 80)
        print(f"SETUP {setup_id} INSPECTION")
        print("=" * 80)
        
        direction = next((e["direction"] for e in events if e.get("direction")), "UNKNOWN").upper()
        final_event = events[-1]
        
        print(f"Direction:     {direction}")
        print(f"Final Event:   {final_event['event']} ({final_event['result']})")
        if final_event.get("reason"):
            print(f"Failure Cause: {final_event['reason']}")
        print("-" * 80)

        # Map timestamps to their events for printing
        event_map = defaultdict(list)
        for e in events:
            event_map[e["timestamp"]].append(f"[{e['event']}: {e['result']}]")

        # Print the candle slice
        print(f"{'TIMESTAMP':<20} | {'OPEN':<8} | {'HIGH':<8} | {'LOW':<8} | {'CLOSE':<8} | {'EVENTS'}")
        print("-" * 80)

        for i in range(slice_start, slice_end + 1):
            candle = self.candles[i]
            
            # Formatting markers
            marker = ">> " if start_idx <= i <= end_idx else "   "
            time_str = candle.timestamp.strftime("%Y-%m-%d %H:%M")
            
            # Gather events for this specific candle
            candle_events = " ".join(event_map.get(candle.timestamp, []))
            
            print(
                f"{marker}{time_str:<17} | "
                f"{candle.open:<8.3f} | {candle.high:<8.3f} | "
                f"{candle.low:<8.3f} | {candle.close:<8.3f} | "
                f"{candle_events}"
            )
            
        print("=" * 80)

import csv
from pathlib import Path

from strategies.sp2l_audit import SP2LAuditEvent


OUTPUT_COLUMNS = [
    "setup_id",
    "event",
    "result",
    "timestamp",
    "state",
    "direction",
    "reason",
    "data",
]


def export_audit_events(
    events: list[SP2LAuditEvent],
    output_path: str | Path,
) -> None:
    output_path = Path(output_path)

    output_path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    with output_path.open(
        "w",
        newline="",
        encoding="utf-8",
    ) as file:
        writer = csv.DictWriter(
            file,
            fieldnames=OUTPUT_COLUMNS,
        )

        writer.writeheader()

        for event in events:
            writer.writerow(
            {
                "setup_id": event.setup_id or "",
                "event": event.event,
                "result": event.result or "",
                "timestamp": event.timestamp.isoformat(),
                "state": event.state,
                "direction": event.direction or "",
                "reason": event.reason or "",
                "data": repr(event.data or {}),
            }
        )

from decimal import Decimal


class Account:
    def __init__(self, starting_balance: str):
        self.starting_balance = Decimal(starting_balance)
        self.balance = self.starting_balance
        self.floating_pnl = Decimal("0")

    @property
    def equity(self) -> Decimal:
        return self.balance + self.floating_pnl

    def apply_pnl(self, pnl: Decimal) -> None:
        self.balance += pnl

    def update_floating_pnl(self, floating_pnl: Decimal) -> None:
        self.floating_pnl = floating_pnl

from dataclasses import dataclass
from datetime import datetime
from decimal import Decimal

from backtest.account import Account
from backtest.risk import RiskManager
from backtest.signal import Signal
from backtest.trade import Trade
from config.instrument import InstrumentConfig
from config.trading import TradingConfig
from models.candle import Candle
from models.price import PriceSpec
from backtest.position_sizing import PositionSizer
from backtest.result import BacktestResult
from backtest.equity import EquityPoint

from strategies.base import Strategy

@dataclass
class PendingOrder:
    signal: Signal
    signal_time: datetime


class BacktestEngine:
    def __init__(
        self,
        candles: list[Candle],
        instrument: InstrumentConfig,
        trading_config: TradingConfig,
        stop_loss_distance: int,
        take_profit_distance: int,
    ):
        self.candles = candles

        self.instrument = instrument
        self.trading_config = trading_config
        self.position_sizer = PositionSizer(trading_config)

        self.price_spec = PriceSpec(
            instrument.tick_size,
            instrument.pip_size,
        )

        self.spread_ticks = self.price_spec.pips_to_ticks(
            trading_config.spread_pips
        )

        self.half_spread_ticks = self.spread_ticks // 2

        self.account = Account(
            str(trading_config.starting_balance)
        )

        self.risk_manager = RiskManager(
            trading_config
        )

        self.stop_loss_distance = stop_loss_distance
        self.take_profit_distance = take_profit_distance

        self.active_trades: list[Trade] = []
        self.completed_trades: list[Trade] = []
        self.open_trades_at_end: list[Trade] = []
        self.final_floating_pnl = Decimal("0")
        self.equity_curve: list[EquityPoint] = []
        self.pending_order: PendingOrder | None = None

    def run(self, strategy: Strategy) -> BacktestResult:
        for candle in self.candles:
            # 1. CANDLE OPENS
            self.process_pending_order(candle)

            # 2. INTRACANDLE MOVEMENT
            self.process_active_trades(candle)

            # 3. EQUITY CURVE
            self.equity_curve.append(
                EquityPoint(
                    timestamp=candle.timestamp,
                    equity=self.account.equity,
                )
            )
            
            # 4. CANDLE CLOSES
            signal = strategy.on_candle(candle)

            # 5. QUEUE SIGNAL FOR NEXT CANDLE
            # CHANGED: Removed the 'self.current_trade is None' blocker
            if signal != Signal.NONE:
                self.pending_order = PendingOrder(
                    signal=signal,
                    signal_time=candle.timestamp,
                )
            else:
                self.pending_order = None
                
        self.finalize_open_trades()
        
        return BacktestResult(
            starting_balance=self.account.starting_balance,
            ending_balance=self.account.balance,
            completed_trades=self.completed_trades,
            open_trades=self.open_trades_at_end,
            final_floating_pnl=self.final_floating_pnl,
            equity_curve=self.equity_curve,
        )

    def process_pending_order(self, candle: Candle) -> None:
        if self.pending_order is None:
            return

        order = self.pending_order
        self.pending_order = None

        # Risk limits are checked at the moment the order
        # would actually execute.
        if not self.risk_manager.can_open_trade(
            candle.timestamp
        ):
            return

        if order.signal == Signal.BUY:
            self.open_buy(candle, order.signal_time)

        elif order.signal == Signal.SELL:
            self.open_sell(candle, order.signal_time)

    def open_buy(
        self,
        candle: Candle,
        signal_time: datetime,
    ) -> None:
        reference_open_ticks = self.price_spec.to_ticks(
            candle.open
        )

        # BUY executes at ASK.
        entry_ticks = (
            reference_open_ticks + self.half_spread_ticks
        )

        stop_loss_ticks = (
            entry_ticks - self.stop_loss_distance
        )

        take_profit_ticks = (
            entry_ticks + self.take_profit_distance
        )

        lot_size = Decimal(
    str(self.trading_config.default_lot)
)

        self.position_sizer.validate_lot_size(lot_size)

        self.active_trades.append(Trade(
            direction="BUY",
            lot_size=lot_size,
            signal_time=signal_time,
            entry_time=candle.timestamp,
            entry_ticks=entry_ticks,
            stop_loss_ticks=stop_loss_ticks,
            take_profit_ticks=take_profit_ticks,
        ))

    def open_sell(
        self,
        candle: Candle,
        signal_time: datetime,
    ) -> None:
        reference_open_ticks = self.price_spec.to_ticks(
            candle.open
        )

        # SELL executes at BID.
        entry_ticks = (
            reference_open_ticks - self.half_spread_ticks
        )

        stop_loss_ticks = (
            entry_ticks + self.stop_loss_distance
        )

        take_profit_ticks = (
            entry_ticks - self.take_profit_distance
        )

        lot_size = Decimal(
            str(self.trading_config.default_lot)
        )

        self.position_sizer.validate_lot_size(lot_size)

        self.active_trades.append(Trade(
            direction="SELL",
            lot_size=lot_size,
            signal_time=signal_time,
            entry_time=candle.timestamp,
            entry_ticks=entry_ticks,
            stop_loss_ticks=stop_loss_ticks,
            take_profit_ticks=take_profit_ticks,
        ))

    def process_active_trades(self, candle: Candle) -> None:
        total_floating = Decimal("0")
        reference_high_ticks = self.price_spec.to_ticks(candle.high)
        reference_low_ticks = self.price_spec.to_ticks(candle.low)

        # Iterate over a copy so we can safely remove closed trades
        for trade in list(self.active_trades):
            closed = False
            if trade.direction == "BUY":
                closed = self.check_buy_exit(candle, trade, reference_low_ticks, reference_high_ticks)
            elif trade.direction == "SELL":
                closed = self.check_sell_exit(candle, trade, reference_low_ticks, reference_high_ticks)
            
            # Only add to floating P&L if it survived the candle
            if not closed:
                floating = self.calculate_trade_floating_pnl(trade, candle)
                total_floating += floating

        self.account.update_floating_pnl(total_floating)
        self.risk_manager.update_floating_pnl(total_floating)



    def check_buy_exit(self, candle: Candle, trade: Trade, reference_low_ticks: int, reference_high_ticks: int) -> bool:
        bid_high_ticks = reference_high_ticks - self.half_spread_ticks
        bid_low_ticks = reference_low_ticks - self.half_spread_ticks

        if bid_low_ticks <= trade.stop_loss_ticks:
            self.close_trade(candle, trade, trade.stop_loss_ticks, "stop_loss")
            return True

        if bid_high_ticks >= trade.take_profit_ticks:
            self.close_trade(candle, trade, trade.take_profit_ticks, "take_profit")
            return True
            
        return False

    def check_sell_exit(self, candle: Candle, trade: Trade, reference_low_ticks: int, reference_high_ticks: int) -> bool:
        ask_high_ticks = reference_high_ticks + self.half_spread_ticks
        ask_low_ticks = reference_low_ticks + self.half_spread_ticks

        if ask_high_ticks >= trade.stop_loss_ticks:
            self.close_trade(candle, trade, trade.stop_loss_ticks, "stop_loss")
            return True

        if ask_low_ticks <= trade.take_profit_ticks:
            self.close_trade(candle, trade, trade.take_profit_ticks, "take_profit")
            return True
            
        return False

    def close_trade(self, candle: Candle, trade: Trade, exit_ticks: int, exit_reason: str) -> None:
        trade.exit_time = candle.timestamp
        trade.exit_ticks = exit_ticks
        trade.exit_reason = exit_reason

        if trade.direction == "BUY":
            trade.profit_ticks = exit_ticks - trade.entry_ticks
        else:
            trade.profit_ticks = trade.entry_ticks - exit_ticks

        price_difference = Decimal(trade.profit_ticks) * self.price_spec.tick_size
        gross_profit = price_difference * Decimal(str(self.instrument.contract_size_oz)) * trade.lot_size
        commission = Decimal(str(self.trading_config.commission_per_lot)) * trade.lot_size
        net_profit = gross_profit - commission

        trade.gross_profit = gross_profit
        trade.commission = commission
        trade.net_profit = net_profit

        self.active_trades.remove(trade)
        self.completed_trades.append(trade)
        
        self.account.apply_pnl(net_profit)
        self.risk_manager.update(trade)

    def calculate_trade_floating_pnl(self, trade: Trade, candle: Candle) -> Decimal:
        reference_price_ticks = self.price_spec.to_ticks(candle.close)

        if trade.direction == "BUY":
            exit_ticks = reference_price_ticks - self.half_spread_ticks
            profit_ticks = exit_ticks - trade.entry_ticks
        else:
            exit_ticks = reference_price_ticks + self.half_spread_ticks
            profit_ticks = trade.entry_ticks - exit_ticks

        price_difference = Decimal(profit_ticks) * self.price_spec.tick_size
        return price_difference * Decimal(str(self.instrument.contract_size_oz)) * trade.lot_size

    def finalize_open_trades(self) -> None:
        self.open_trades_at_end = list(self.active_trades)

        if not self.candles or not self.active_trades:
            self.final_floating_pnl = Decimal("0")
            return

        final_candle = self.candles[-1]
        
        total_floating = sum(
            self.calculate_trade_floating_pnl(trade, final_candle)
            for trade in self.active_trades
        )

        self.final_floating_pnl = total_floating
        self.account.update_floating_pnl(total_floating)
        self.risk_manager.update_floating_pnl(total_floating)

class BacktestMetrics:

class MetricsCalculator:
    def calculate(self, result: BacktestResult) -> BacktestMetrics:
        trades = result.completed_trades
        total_trades = len(trades)

        winning_trades = sum(
            1
            for trade in trades
            if trade.net_profit is not None
            and trade.net_profit > 0
        )

        losing_trades = sum(
            1
            for trade in trades
            if trade.net_profit is not None
            and trade.net_profit < 0
        )

        # Trade profit lists
        profits = [
            trade.net_profit
            for trade in trades
            if trade.net_profit is not None
            and trade.net_profit > 0
        ]

        losses = [
            trade.net_profit
            for trade in trades
            if trade.net_profit is not None
            and trade.net_profit < 0
        ]

        # Win rate
        if total_trades > 0:
            win_rate_percent = (
                Decimal(winning_trades)
                / Decimal(total_trades)
                * Decimal("100")
            ).quantize(
                Decimal("0.01"),
                rounding=ROUND_HALF_UP,
            )
        else:
            win_rate_percent = Decimal("0.00")

        # Net profit and return
        net_profit = result.net_profit

        if result.starting_balance != 0:
            return_percent = (
                net_profit
                / result.starting_balance
                * Decimal("100")
            ).quantize(
                Decimal("0.01"),
                rounding=ROUND_HALF_UP,
            )
        else:
            return_percent = Decimal("0.00")

        # Average trade
        if total_trades > 0:
            total_trade_profit = sum(
                (
                    trade.net_profit
                    for trade in trades
                    if trade.net_profit is not None
                ),
                Decimal("0"),
            )

            average_trade = (
                total_trade_profit
                / Decimal(total_trades)
            ).quantize(
                Decimal("0.001"),
                rounding=ROUND_HALF_UP,
            )
        else:
            average_trade = Decimal("0.000")

        # Profit factor
        if losses:
            gross_profit = sum(profits, Decimal("0"))
            gross_loss = sum(losses, Decimal("0"))

            profit_factor = (
                gross_profit / abs(gross_loss)
            ).quantize(
                Decimal("0.01"),
                rounding=ROUND_HALF_UP,
            )
        else:
            profit_factor = Decimal("0.00")

        # Largest win / loss
        largest_win = max(
            profits,
            default=Decimal("0"),
        )

        largest_loss = min(
            losses,
            default=Decimal("0"),
        )

        # Maximum consecutive losses
        max_consecutive_losses = 0
        current_consecutive_losses = 0

        for trade in trades:
            if (
                trade.net_profit is not None
                and trade.net_profit < 0
            ):
                current_consecutive_losses += 1

                max_consecutive_losses = max(
                    max_consecutive_losses,
                    current_consecutive_losses,
                )
            else:
                current_consecutive_losses = 0

        # Maximum drawdown
        max_drawdown = Decimal("0")
        max_drawdown_percent = Decimal("0.00")

        peak_equity = None

        for point in result.equity_curve:
            if (
                peak_equity is None
                or point.equity > peak_equity
            ):
                peak_equity = point.equity

            drawdown = peak_equity - point.equity

            if drawdown > max_drawdown:
                max_drawdown = drawdown

                if peak_equity != 0:
                    max_drawdown_percent = (
                        drawdown
                        / peak_equity
                        * Decimal("100")
                    ).quantize(
                        Decimal("0.01"),
                        rounding=ROUND_HALF_UP,
                    )
                # Market exposure
        total_time_in_market = timedelta(0)
        exposure_percent = Decimal("0.00")

        if result.equity_curve:
            backtest_start = result.equity_curve[0].timestamp
            backtest_end = result.equity_curve[-1].timestamp
            backtest_duration = backtest_end - backtest_start

            for trade in trades:
                if trade.exit_time is not None:
                    total_time_in_market += (
                        trade.exit_time - trade.entry_time
                    )

            if result.open_trades:
                for trade in result.open_trades:
                    total_time_in_market += (
                        backtest_end - trade.entry_time
                    )

            if backtest_duration.total_seconds() > 0:
                exposure_percent = (
                    Decimal(str(
                        total_time_in_market.total_seconds()
                    ))
                    / Decimal(str(
                        backtest_duration.total_seconds()
                    ))
                    * Decimal("100")
                ).quantize(
                    Decimal("0.01"),
                    rounding=ROUND_HALF_UP,
                )

        return BacktestMetrics(
            starting_balance=result.starting_balance,
            ending_balance=result.ending_balance,
            net_profit=net_profit,
            return_percent=return_percent,
            total_trades=total_trades,
            winning_trades=winning_trades,
            losing_trades=losing_trades,
            max_consecutive_losses=max_consecutive_losses,
            win_rate_percent=win_rate_percent,
            profit_factor=profit_factor,
            average_trade=average_trade,
            largest_win=largest_win,
            largest_loss=largest_loss,
            max_drawdown=max_drawdown,
            max_drawdown_percent=max_drawdown_percent,
            total_time_in_market=total_time_in_market,
            exposure_percent=exposure_percent,
        )

from config.trading import TradingConfig


class PositionSizer:
    def __init__(self, trading_config: TradingConfig):
        self.minimum_lot = Decimal(str(trading_config.minimum_lot))
        self.lot_step = Decimal(str(trading_config.lot_step))

    def validate_lot_size(self, lot_size: Decimal) -> None:
        if lot_size < self.minimum_lot:
            raise ValueError(
                f"Lot size {lot_size} is below the minimum "
                f"of {self.minimum_lot}"
            )

        offset = lot_size - self.minimum_lot

        if offset % self.lot_step != 0:
            raise ValueError(
                f"Lot size {lot_size} is not a valid lot step. "
                f"Minimum: {self.minimum_lot}, "
                f"step: {self.lot_step}"
            )

    def normalize_lot_size(self, lot_size: Decimal) -> Decimal:
        if lot_size < self.minimum_lot:
            return self.minimum_lot

        steps = (
            (lot_size - self.minimum_lot) / self.lot_step
        ).to_integral_value()

        normalized = (
            self.minimum_lot
            + steps * self.lot_step
        )

        self.validate_lot_size(normalized)
        return normalized

from datetime import date, datetime
from decimal import Decimal

from backtest.trade import Trade
from config.trading import TradingConfig


class RiskManager:
    def __init__(self, trading_config: TradingConfig):
        self.trading_config = trading_config

        self.daily_realized_pnl = Decimal("0")
        self.weekly_realized_pnl = Decimal("0")

        self.floating_pnl = Decimal("0")

        self.current_day: date | None = None
        self.current_week: tuple[int, int] | None = None

    def update(self, trade: Trade) -> None:
        """
        Add a completed trade's net P&L to the current
        daily and weekly totals.
        """

        if trade.exit_time is None:
            return

        if trade.net_profit is None:
            return

        self._update_periods(trade.exit_time)

        self.daily_realized_pnl += trade.net_profit
        self.weekly_realized_pnl += trade.net_profit

    def can_open_trade(self, timestamp: datetime) -> bool:
        """
        Return True if the daily and weekly loss limits
        have not been reached.
        """

        self._update_periods(timestamp)

        if self.daily_loss_reached():
            return False

        if self.weekly_loss_reached():
            return False

        return True

    def daily_loss_reached(self) -> bool:
        return (
            self.current_daily_pnl()
            <= -Decimal(
                str(self.trading_config.maximum_daily_loss)
            )
        )

    def weekly_loss_reached(self) -> bool:
        return (
            self.current_weekly_pnl()
            <= -Decimal(
                str(self.trading_config.maximum_weekly_loss)
            )
        )

    def _update_periods(self, timestamp: datetime) -> None:
        current_day = timestamp.date()

        iso_calendar = timestamp.isocalendar()
        current_week = (
            iso_calendar.year,
            iso_calendar.week,
        )

        if self.current_day != current_day:
            self.current_day = current_day
            self.daily_realized_pnl = Decimal("0")

        if self.current_week != current_week:
            self.current_week = current_week
            self.weekly_realized_pnl = Decimal("0")

    def update_floating_pnl(self, floating_pnl: Decimal) -> None:
        self.floating_pnl = floating_pnl

    def current_daily_pnl(self) -> Decimal:
        return self.daily_realized_pnl + self.floating_pnl

    def current_weekly_pnl(self) -> Decimal:
        return self.weekly_realized_pnl + self.floating_pnl

import csv
from datetime import datetime
from pathlib import Path

from models.candle import Candle


class DataLoadError(Exception):
    pass


def load_dataset(path: Path, dataset_name: str) -> list[Candle]:
    if not path.exists():
        raise DataLoadError(f"{dataset_name}: file not found: {path}")

    candles: list[Candle] = []

    with path.open("r", encoding="utf-8-sig", newline="") as file:
        reader = csv.reader(file)

        try:
            header = next(reader)
        except StopIteration:
            raise DataLoadError(f"{dataset_name}: file is empty")

        header = [column.strip().lower() for column in header]

        for line_number, row in enumerate(reader, start=2):
            if not row or all(not value.strip() for value in row):
                continue

            try:
                timestamp, open_price, high_price, low_price, close_price = (
                    parse_row(row)
                )

            except (ValueError, IndexError) as exc:
                raise DataLoadError(
                    f"{dataset_name}: invalid data on line "
                    f"{line_number}: {row}"
                ) from exc

            validate_ohlc(
                dataset_name,
                line_number,
                timestamp,
                open_price,
                high_price,
                low_price,
                close_price,
            )

            candles.append(
                Candle(
                    timestamp=timestamp,
                    open=open_price,
                    high=high_price,
                    low=low_price,
                    close=close_price,
                )
            )

    if not candles:
        raise DataLoadError(f"{dataset_name}: no candles loaded")

    validate_order(dataset_name, candles)

    return candles


def parse_row(
    row: list[str],
) -> tuple[datetime, float, float, float, float]:

    # Dataset A
    #
    # date and time, open, high, low, close, volume
    #
    # Example:
    # 2023-11-22 09:20,1999.905,2000.005,1999.015,1999.485,5

    if len(row) == 6:
        timestamp = datetime.strptime(
            row[0].strip(),
            "%Y-%m-%d %H:%M",
        )

        open_price = float(row[1])
        high_price = float(row[2])
        low_price = float(row[3])
        close_price = float(row[4])

        return (
            timestamp,
            open_price,
            high_price,
            low_price,
            close_price,
        )

    # Dataset B
    #
    # date, time, open, high, low, close, volume
    #
    # Example:
    # 2026.09.15,09:05,4291.76,4291.76,4286.10,4288.71,908

    if len(row) == 7:
        timestamp = datetime.strptime(
            f"{row[0].strip()} {row[1].strip()}",
            "%Y.%m.%d %H:%M",
        )

        open_price = float(row[2])
        high_price = float(row[3])
        low_price = float(row[4])
        close_price = float(row[5])

        return (
            timestamp,
            open_price,
            high_price,
            low_price,
            close_price,
        )

    raise ValueError(
        f"Unexpected number of columns: {len(row)}"
    )


def validate_ohlc(
    dataset_name: str,
    line_number: int,
    timestamp: datetime,
    open_price: float,
    high_price: float,
    low_price: float,
    close_price: float,
) -> None:

    if high_price < max(open_price, close_price):
        raise DataLoadError(
            f"{dataset_name}: line {line_number}: "
            f"high is lower than open/close"
        )

    if low_price > min(open_price, close_price):
        raise DataLoadError(
            f"{dataset_name}: line {line_number}: "
            f"low is higher than open/close"
        )

    if low_price > high_price:
        raise DataLoadError(
            f"{dataset_name}: line {line_number}: "
            f"low is greater than high"
        )


def validate_order(
    dataset_name: str,
    candles: list[Candle],
) -> None:

    previous_timestamp = None

    for candle in candles:
        if previous_timestamp is not None:
            if candle.timestamp <= previous_timestamp:
                raise DataLoadError(
                    f"{dataset_name}: timestamps are not strictly "
                    f"increasing around {candle.timestamp}"
                )

        previous_timestamp = candle.timestamp

from decimal import Decimal, ROUND_HALF_UP


class PriceSpec:
    def __init__(self, tick_size: str, pip_size: str):
        self.tick_size = Decimal(tick_size)
        self.pip_size = Decimal(pip_size)

    def to_ticks(self, price: float | str | Decimal) -> int:
        value = Decimal(str(price))

        return int(
            (value / self.tick_size).quantize(
                Decimal("1"),
                rounding=ROUND_HALF_UP,
            )
        )

    def from_ticks(self, ticks: int) -> Decimal:
        return Decimal(ticks) * self.tick_size

    def pips_to_ticks(
        self,
        pips: Decimal | float | str,
    ) -> int:
        price_distance = Decimal(str(pips)) * self.pip_size

        return int(
            (price_distance / self.tick_size).quantize(
                Decimal("1"),
                rounding=ROUND_HALF_UP,
            )
        )

    def format_price(self, ticks: int) -> str:
        return format(self.from_ticks(ticks), "f")

from collections import deque
from datetime import timedelta

from models.candle import Candle


class M15Builder:
    def __init__(self, max_history: int = 20):
        if max_history <= 0:
            raise ValueError("max_history must be greater than zero")

        self.max_history = max_history
        self.history: deque[Candle] = deque(maxlen=max_history)
        self._buffer: list[Candle] = []

    def update(self, candle: Candle) -> Candle | None:
        # No active group: only an M15-aligned candle can start one.
        if not self._buffer:
            if candle.timestamp.minute % 15 != 0:
                return None

            self._buffer.append(candle)
            return None

        # We already have the first candle of the group.
        expected_timestamp = (
            self._buffer[-1].timestamp + timedelta(minutes=5)
        )

        # Any unexpected timestamp invalidates the current group.
        if candle.timestamp != expected_timestamp:
            self._buffer = []

            # The current candle may itself be a valid new M15 starting point.
            if candle.timestamp.minute % 15 == 0:
                self._buffer.append(candle)

            return None

        self._buffer.append(candle)

        # We need exactly three consecutive M5 candles.
        if len(self._buffer) < 3:
            return None

        candles = self._buffer
        self._buffer = []

        m15 = Candle(
            timestamp=candles[0].timestamp,
            open=candles[0].open,
            high=max(c.high for c in candles),
            low=min(c.low for c in candles),
            close=candles[-1].close,
        )

        self.history.append(m15)

        return m15

    @property
    def candles(self) -> tuple[Candle, ...]:
        return tuple(self.history)

from models.candle import Candle
from strategies.signal_bar import SignalBarEvent

@dataclass(frozen=True)
class KeyBarEvent:
    direction: str
    signal_bar: SignalBarEvent
    key_candle: Candle

class KeyBarDetector:
    """
    Detects the candle immediately following the Signal Bar.

    ```
    Bullish:
        - Key candle breaks Signal Bar high.
        - Key candle closes above Signal Bar high.

    Bearish:
        - Key candle breaks Signal Bar low.
        - Key candle closes below Signal Bar low.

    The detector only confirms the setup.
    It does not execute a trade.
    """

    def detect(
        self,
        signal_bar: SignalBarEvent,
        key_candle: Candle,
    ) -> KeyBarEvent | None:

        signal_candle = signal_bar.signal_candle

        expected_timestamp = (
            signal_candle.timestamp + timedelta(minutes=5)
        )

        # The Key Bar must be the immediately following M5 candle.
        # Any gap or later candle invalidates this Signal Bar.
        if key_candle.timestamp != expected_timestamp:
            return None

        if signal_bar.direction == "buy":
            if (
                key_candle.high > signal_candle.high
                and key_candle.close > signal_candle.high
            ):
                return KeyBarEvent(
                    direction="buy",
                    signal_bar=signal_bar,
                    key_candle=key_candle,
                )

        elif signal_bar.direction == "sell":
            if (
                key_candle.low < signal_candle.low
                and key_candle.close < signal_candle.low
            ):
                return KeyBarEvent(
                    direction="sell",
                    signal_bar=signal_bar,
                    key_candle=key_candle,
                )

        return None

from models.candle import Candle


@dataclass(frozen=True)
class PullbackStructureEvent:
    direction: str
    second_leg: SecondLegEvent
    anchor_candle: Candle
    structure_points: list[StructurePoint]


class PullbackStructureDetector:
    """
    Detects the counter-trend structure sequence inside the SP2L
    second-leg pullback.

    The Second Leg pullback candle is the structural anchor.

    It is NOT counted as structure #1.

    Bullish setup:
        - Second Leg candle is the anchor high.
        - Subsequent lower swing highs form LH #1, LH #2, ...
        - At least 3 consecutive Lower Highs are required.
        - At most 7 consecutive Lower Highs are allowed.
        - An 8th consecutive Lower High invalidates the setup.

    Bearish setup:
        - Second Leg candle is the anchor low.
        - Subsequent higher swing lows form HL #1, HL #2, ...
        - At least 3 consecutive Higher Lows are required.
        - At most 7 consecutive Higher Lows are allowed.
        - An 8th consecutive Higher Low invalidates the setup.

    If the structural condition is broken, the sequence resets and a
    new sequence may begin from the next structural point.

    The detector does not determine:
        - Signal Bar
        - Key Bar
        - entry
        - stop loss
    """

    def __init__(
        self,
        history: SwingHistory,
        minimum_length: int = 3,
        maximum_length: int = 7,
    ):
        if minimum_length <= 0:
            raise ValueError("minimum_length must be greater than zero")

        if maximum_length < minimum_length:
            raise ValueError(
                "maximum_length must be greater than or equal to "
                "minimum_length"
            )

        self.history = history
        self.minimum_length = minimum_length
        self.maximum_length = maximum_length

    def detect(
        self,
        second_leg: SecondLegEvent,
    ) -> PullbackStructureEvent | None:

        anchor_candle = second_leg.pullback_candle

        if second_leg.direction == "buy":
            sequence = self._lower_high_sequence(anchor_candle)

        elif second_leg.direction == "sell":
            sequence = self._higher_low_sequence(anchor_candle)

        else:
            return None

        if len(sequence) < self.minimum_length:
            return None

        if len(sequence) > self.maximum_length:
            return None

        return PullbackStructureEvent(
            direction=second_leg.direction,
            second_leg=second_leg,
            anchor_candle=anchor_candle,
            structure_points=sequence,
        )

    def _higher_low_sequence(self, anchor_candle):
        anchor_low = Decimal(str(anchor_candle.low))

        lows = [
            point
            for point in self.history.lows()
            if point.candle.timestamp > anchor_candle.timestamp
        ]

        sequence: list[StructurePoint] = []
        previous_price = anchor_low

        for point in lows:
            # Breaking the original Anchor permanently invalidates
            # the entire setup.
            if point.price < anchor_low:
                return []

            if point.price > previous_price:
                sequence.append(
                    StructurePoint(
                        type="higher_low",
                        candle=point.candle,
                        price=point.price,
                    )
                )
                previous_price = point.price

                # An 8th consecutive HL invalidates only the
                # current compression sequence. Start fresh
                # from this structural point.
                if len(sequence) > self.maximum_length:
                    sequence = []
                    previous_price = point.price

            else:
                # Structural violation, but the Anchor is still intact.
                # Start a fresh sequence from this structural point.
                sequence = []
                previous_price = point.price

        return sequence

    def _lower_high_sequence(self, anchor_candle):
        anchor_high = Decimal(str(anchor_candle.high))

        highs = [
            point
            for point in self.history.highs()
            if point.candle.timestamp > anchor_candle.timestamp
        ]

        sequence: list[StructurePoint] = []
        previous_price = anchor_high

        for point in highs:
            # Breaking the original Anchor permanently invalidates
            # the entire setup.
            if point.price > anchor_high:
                return []

            if point.price < previous_price:
                sequence.append(
                    StructurePoint(
                        type="lower_high",
                        candle=point.candle,
                        price=point.price,
                    )
                )
                previous_price = point.price

                # An 8th consecutive LH invalidates only the
                # current compression sequence. Start fresh
                # from this structural point.
                if len(sequence) > self.maximum_length:
                    sequence = []
                    previous_price = point.price

            else:
                # Structural violation, but the Anchor is still intact.
                # Start a fresh sequence from this structural point.
                sequence = []
                previous_price = point.price
        
        return sequence

from dataclasses import dataclass

from models.candle import Candle
from strategies.spike_event import SpikeEvent


@dataclass(frozen=True)
class PGapEvent:
    direction: str
    pre_breakout_candle: Candle
    breakout_candle: Candle
    follow_through_candle: Candle


class PGapDetector:
    """
    Detects a strict non-overlapping pressure gap (P-Gap).

    Source-derived rule:
        After a breakout, subsequent follow-through price action must
        not pull back far enough for its shadow to overlap the shadow
        of the preceding pre-breakout candle.

    Current implementation assumption:
        The first candle immediately after the Spike is treated as the
        follow-through candidate.

    The Spike itself is supplied as a SpikeEvent, so this detector does
    not independently determine whether the breakout candle is a Spike.
    """

    def detect(
        self,
        spike_event: SpikeEvent,
        pre_breakout_candle: Candle,
        follow_through_candle: Candle,
    ) -> PGapEvent | None:

        if spike_event.direction == "buy":
            if follow_through_candle.low > pre_breakout_candle.high:
                return PGapEvent(
                    direction="buy",
                    pre_breakout_candle=pre_breakout_candle,
                    breakout_candle=spike_event.breakout_candle,
                    follow_through_candle=follow_through_candle,
                )

        if spike_event.direction == "sell":
            if follow_through_candle.high < pre_breakout_candle.low:
                return PGapEvent(
                    direction="sell",
                    pre_breakout_candle=pre_breakout_candle,
                    breakout_candle=spike_event.breakout_candle,
                    follow_through_candle=follow_through_candle,
                )

        return None

from dataclasses import dataclass
from decimal import Decimal

from models.candle import Candle
from strategies.p_gap import PGapEvent


@dataclass(frozen=True)
class SecondLegEvent:
    direction: str
    p_gap: PGapEvent
    pullback_candle: Candle
    reference_candle: Candle


class SecondLegDetector:
    """
    Detects the source-defined second-leg/pullback condition.

    Bullish setup:
        The corrective candle reaches the low of the previous candle.

    Bearish setup:
        The corrective candle reaches the high of the previous candle.

    This detector deliberately does not determine:
        - how many pullback candles exist
        - the 3-7 lower-high / higher-low sequence
        - Signal Bar
        - Key Bar
        - entry

    Those are separate parts of the strategy.
    """

    def detect(
        self,
        p_gap_event: PGapEvent,
        reference_candle: Candle,
        pullback_candle: Candle,
    ) -> SecondLegEvent | None:

        if p_gap_event.direction == "buy":
            if Decimal(str(pullback_candle.low)) <= Decimal(
                str(reference_candle.low)
            ):
                return SecondLegEvent(
                    direction="buy",
                    p_gap=p_gap_event,
                    pullback_candle=pullback_candle,
                    reference_candle=reference_candle,
                )

        if p_gap_event.direction == "sell":
            if Decimal(str(pullback_candle.high)) >= Decimal(
                str(reference_candle.high)
            ):
                return SecondLegEvent(
                    direction="sell",
                    p_gap=p_gap_event,
                    pullback_candle=pullback_candle,
                    reference_candle=reference_candle,
                )

        return None

from dataclasses import dataclass

from models.candle import Candle
from strategies.pullback_structure import PullbackStructureEvent


@dataclass(frozen=True)
class SignalBarEvent:
    direction: str
    pullback_structure: PullbackStructureEvent
    signal_candle: Candle


class SignalBarDetector:
    """
    Detects the SP2L Signal Bar after the corrective structure sequence.

    Bullish:
        The signal candle breaks the previous corrective candle's high.

    Bearish:
        The signal candle breaks the previous corrective candle's low.

    The Signal Bar itself does NOT trigger entry.
    The following candle must qualify as the Key Bar.
    """

    def detect(
        self,
        pullback_structure: PullbackStructureEvent,
        previous_candle: Candle,
        signal_candle: Candle,
    ) -> SignalBarEvent | None:

        if signal_candle.timestamp <= previous_candle.timestamp:
            return None

        if pullback_structure.direction == "buy":
            if self._bullish_break_of_previous_high(
                previous_candle,
                signal_candle,
            ):
                return SignalBarEvent(
                    direction="buy",
                    pullback_structure=pullback_structure,
                    signal_candle=signal_candle,
                )

        elif pullback_structure.direction == "sell":
            if self._bearish_break_of_previous_low(
                previous_candle,
                signal_candle,
            ):
                return SignalBarEvent(
                    direction="sell",
                    pullback_structure=pullback_structure,
                    signal_candle=signal_candle,
                )

        return None

    @staticmethod
    def _bullish_break_of_previous_high(
        previous_candle: Candle,
        signal_candle: Candle,
    ) -> bool:
        return signal_candle.high > previous_candle.high

    @staticmethod
    def _bearish_break_of_previous_low(
        previous_candle: Candle,
        signal_candle: Candle,
    ) -> bool:
        return signal_candle.low < previous_candle.low

from collections import deque

from models.candle import Candle
from strategies.key_bar import KeyBarDetector
from strategies.p_gap import PGapDetector
from strategies.pullback_structure import PullbackStructureDetector
from strategies.second_leg import SecondLegDetector
from strategies.signal_bar import SignalBarDetector
from strategies.sp2l_state import SP2LSetup, SP2LState
from strategies.spike import SpikeDetector
from strategies.spike_event import SpikeEvent
from strategies.swing_processor import SwingProcessor

from strategies.sp2l_audit import SP2LAuditRecorder

class SP2LStateMachine:
    """
    Causal SP2L strategy state machine.

    Processes one completed candle at a time.

    State flow:

        IDLE
          ↓
        SPIKE_VALIDATED
          ↓
        PULLBACK
          ↓
        COMPRESSION_VALID
          ↓
        WAITING_FOR_KEY_BAR
          ↓
        TRIGGER

    The machine only recognizes setups.
    Trade execution is handled separately.
    """

    def __init__(
        self,
        spike_detector: SpikeDetector | None = None,
        p_gap_detector: PGapDetector | None = None,
        second_leg_detector: SecondLegDetector | None = None,
        pullback_structure_detector: PullbackStructureDetector | None = None,
        signal_bar_detector: SignalBarDetector | None = None,
        key_bar_detector: KeyBarDetector | None = None,
        swing_processor: SwingProcessor | None = None,
        audit_recorder: SP2LAuditRecorder | None = None,
        
    ):
        self.audit_recorder = audit_recorder
        self._setup_id: int | None = None
        self.spike_detector = spike_detector or SpikeDetector()
        self.p_gap_detector = p_gap_detector or PGapDetector()
        self.second_leg_detector = (
            second_leg_detector or SecondLegDetector()
        )
        self.signal_bar_detector = (
            signal_bar_detector or SignalBarDetector()
        )
        self.key_bar_detector = key_bar_detector or KeyBarDetector()

        self.swing_processor = swing_processor or SwingProcessor()
        if pullback_structure_detector is None:
            pullback_structure_detector = PullbackStructureDetector(
                history=self.swing_processor.history
            )

        self.pullback_structure_detector = pullback_structure_detector

        self._spike_lookback = self.spike_detector.config.lookback

        self._candles = deque(
            maxlen=self._spike_lookback + 2
        )

        self.state = SP2LState.IDLE

        self._previous_candle: Candle | None = None
        self._pre_breakout_candle: Candle | None = None

        self._pending_spike: SpikeEvent | None = None
        self._p_gap = None
        self._second_leg = None
        self._pullback_structure = None
        self._signal_bar = None

    @property
    def pending_spike(self) -> SpikeEvent | None:
        return self._pending_spike

    @property
    def p_gap(self):
        return self._p_gap

    @property
    def second_leg(self):
        return self._second_leg

    @property
    def pullback_structure(self):
        return self._pullback_structure

    @property
    def signal_bar(self):
        return self._signal_bar

    def process(self, candle: Candle) -> SP2LSetup | None:
        """
        Process one completed candle.

        Returns:
            SP2LSetup when a Key Bar confirms the setup.
            None otherwise.
        """
        self._candles.append(candle)

        self.swing_processor.process(candle)

        if self.state == SP2LState.IDLE:
            return self._process_idle(candle)

        if self.state == SP2LState.SPIKE_VALIDATED:
            return self._process_spike_validated(candle)

        if self.state == SP2LState.PULLBACK:
            return self._process_pullback(candle)

        if self.state == SP2LState.COMPRESSION_VALID:
            return self._process_compression(candle)

        if self.state == SP2LState.WAITING_FOR_KEY_BAR:
            return self._process_key_bar(candle)

        self._reset()
        self._previous_candle = candle

        return None

    def _process_idle(
        self,
        candle: Candle,
    ) -> SP2LSetup | None:
        required_candles = self._spike_lookback + 1

        if len(self._candles) < required_candles:
            self._previous_candle = candle
            return None

        spike = self.spike_detector.detect_event(
            list(self._candles)
        )

        if spike is not None:
            self._pending_spike = spike
            self._pre_breakout_candle = self._candles[-2]
            self.state = SP2LState.SPIKE_VALIDATED

            if self.audit_recorder is not None:
                self._setup_id = self.audit_recorder.new_setup_id()
                self.audit_recorder.record(
                    event="SPIKE",
                    timestamp=spike.breakout_candle.timestamp,
                    state=self.state.value,
                    setup_id=self._setup_id,
                    direction=spike.direction,
                    result="SUCCESS",
                    data={
                        "breakout_timestamp": (
                            spike.breakout_candle.timestamp
                        ),
                        "pre_breakout_timestamp": (
                            self._pre_breakout_candle.timestamp
                        ),
                    },
                )

        self._previous_candle = candle

        return None

    def _process_spike_validated(
        self,
        candle: Candle,
    ) -> SP2LSetup | None:
        spike = self._pending_spike
        pre_breakout_candle = self._pre_breakout_candle

        if spike is None or pre_breakout_candle is None:
            self._reset()
            self._previous_candle = candle
            return None

        if candle.timestamp <= spike.breakout_candle.timestamp:
            self._previous_candle = candle
            return None

        p_gap = self.p_gap_detector.detect(
            spike_event=spike,
            pre_breakout_candle=pre_breakout_candle,
            follow_through_candle=candle,
        )

        if p_gap is None:
            if self.audit_recorder is not None:
                self.audit_recorder.record(
                    event="P_GAP",
                    timestamp=candle.timestamp,
                    state=self.state.value,
                    setup_id=self._setup_id,
                    direction=spike.direction,
                    result="FAILED",
                    reason="follow_through_candle_did_not_form_p_gap",
                    data={
                        "spike_timestamp": (
                            spike.breakout_candle.timestamp
                        ),
                        "pre_breakout_timestamp": (
                            pre_breakout_candle.timestamp
                        ),
                        "follow_through_timestamp": (
                            candle.timestamp
                        ),
                    },
                )

            self._reset()
            self._previous_candle = candle
            return None

        if self.audit_recorder is not None:
            self.audit_recorder.record(
                event="P_GAP",
                timestamp=candle.timestamp,
                state=SP2LState.PULLBACK.value,
                setup_id=self._setup_id,
                direction=p_gap.direction,
                result="SUCCESS",
                data={
                    "spike_timestamp": (
                        spike.breakout_candle.timestamp
                    ),
                    "pre_breakout_timestamp": (
                        pre_breakout_candle.timestamp
                    ),
                    "follow_through_timestamp": (
                        candle.timestamp
                    ),
                },
            )

        self._p_gap = p_gap
        self._previous_candle = candle
        self.state = SP2LState.PULLBACK

        return None

    def _process_pullback(
        self,
        candle: Candle,
    ) -> SP2LSetup | None:
        if self._p_gap is None:
            self._reset()
            self._previous_candle = candle
            return None

        if self._second_leg is None:
            reference_candle = self._previous_candle

            if reference_candle is None:
                self._previous_candle = candle
                return None

            second_leg = self.second_leg_detector.detect(
                p_gap_event=self._p_gap,
                reference_candle=reference_candle,
                pullback_candle=candle,
            )

            if second_leg is None:
                self._previous_candle = candle
                return None

            self._second_leg = second_leg

            if self.audit_recorder is not None:
                self.audit_recorder.record(
                    event="ANCHOR",
                    timestamp=candle.timestamp,
                    state=SP2LState.PULLBACK.value,
                    setup_id=self._setup_id,
                    direction=second_leg.direction,
                    result="SUCCESS",
                    data={
                        "p_gap_timestamp": (
                            self._p_gap.follow_through_candle.timestamp
                        ),
                        "reference_timestamp": (
                            reference_candle.timestamp
                        ),
                        "anchor_timestamp": (
                            second_leg.pullback_candle.timestamp
                        ),
                    },
                )
            self._previous_candle = candle

            return None

        if self._anchor_invalidated(candle):
            self._reset()
            self._previous_candle = candle
            return None

        pullback_structure = (
            self.pullback_structure_detector.detect(
                self._second_leg
            )
        )

        if pullback_structure is not None:
            self._pullback_structure = pullback_structure
            self.state = SP2LState.COMPRESSION_VALID

            if self.audit_recorder is not None:
                self.audit_recorder.record(
                    event="COMPRESSION",
                    timestamp=candle.timestamp,
                    state=self.state.value,
                    setup_id=self._setup_id,
                    direction=pullback_structure.direction,
                    result="SUCCESS",
                    data={
                        "anchor_timestamp": (
                            pullback_structure.anchor_candle.timestamp
                        ),
                        "structure_count": len(
                            pullback_structure.structure_points
                        ),
                        "structure_timestamps": [
                            point.candle.timestamp
                            for point in (
                                pullback_structure.structure_points
                            )
                        ],
                    },
                )

        self._previous_candle = candle

        return None

    def _process_compression(
        self,
        candle: Candle,
    ) -> SP2LSetup | None:
        if self._second_leg is None:
            self._reset()
            self._previous_candle = candle
            return None

        if self._pullback_structure is None:
            self.state = SP2LState.PULLBACK
            self._previous_candle = candle
            return None

        if self._anchor_invalidated(candle):
            self._reset()
            self._previous_candle = candle
            return None

        # Re-check the corrective structure before looking
        # for a Signal Bar. A structural violation can invalidate
        # the current compression sequence without invalidating
        # the original Anchor.
        pullback_structure = (
            self.pullback_structure_detector.detect(
                self._second_leg
            )
        )

        if pullback_structure is None:
            self._pullback_structure = None
            self.state = SP2LState.PULLBACK
            self._previous_candle = candle
            return None

        self._pullback_structure = pullback_structure

        previous_candle = self._previous_candle

        if previous_candle is None:
            self._previous_candle = candle
            return None

        signal_bar = self.signal_bar_detector.detect(
            pullback_structure=self._pullback_structure,
            previous_candle=previous_candle,
            signal_candle=candle,
        )

        if signal_bar is not None:
            self._signal_bar = signal_bar
            self.state = SP2LState.WAITING_FOR_KEY_BAR

            if self.audit_recorder is not None:
                self.audit_recorder.record(
                    event="SIGNAL_BAR",
                    timestamp=candle.timestamp,
                    state=self.state.value,
                    setup_id=self._setup_id,
                    direction=signal_bar.direction,
                    result="SUCCESS",
                    data={
                        "signal_timestamp": (
                            signal_bar.signal_candle.timestamp
                        ),
                        "compression_timestamp": (
                            self._pullback_structure
                            .structure_points[-1]
                            .candle.timestamp
                        ),
                        "structure_count": len(
                            self._pullback_structure.structure_points
                        ),
                    },
                )

        self._previous_candle = candle

        return None

    def _process_key_bar(
        self,
        candle: Candle,
    ) -> SP2LSetup | None:
        if self._signal_bar is None:
            self._reset()
            self._previous_candle = candle
            return None

        key_bar = self.key_bar_detector.detect(
    signal_bar=self._signal_bar,
    key_candle=candle,
)

        if key_bar is None:
            if self.audit_recorder is not None:
                self.audit_recorder.record(
                    event="KEY_BAR",
                    timestamp=candle.timestamp,
                    state=self.state.value,
                    setup_id=self._setup_id,
                    direction=self._signal_bar.direction,
                    result="FAILED",
                    reason="key_bar_condition_not_met",
                    data={
                        "signal_timestamp": (
                            self._signal_bar.signal_candle.timestamp
                        ),
                        "key_candidate_timestamp": (
                            candle.timestamp
                        ),
                    },
                )

            self._reset()
            self._previous_candle = candle
            return None

        if self.audit_recorder is not None:
            self.audit_recorder.record(
                event="KEY_BAR",
                timestamp=candle.timestamp,
                state=self.state.value,
                setup_id=self._setup_id,
                direction=key_bar.direction,
                result="SUCCESS",
                data={
                    "signal_timestamp": (
                        key_bar.signal_bar.signal_candle.timestamp
                    ),
                    "key_timestamp": (
                        key_bar.key_candle.timestamp
                    ),
                },
            )

        if (
            self._pending_spike is None
            or self._p_gap is None
            or self._pullback_structure is None
        ):
            self._reset()
            self._previous_candle = candle
            return None

        setup = SP2LSetup(
            direction=key_bar.direction,
            spike=self._pending_spike,
            p_gap=self._p_gap,
            pullback_structure=self._pullback_structure,
            signal_bar=self._signal_bar,
            key_bar=key_bar,
        )

        self._reset()
        self._previous_candle = candle

        return setup

    def _anchor_invalidated(
        self,
        candle: Candle,
    ) -> bool:
        if self._second_leg is None:
            return False

        anchor = self._second_leg.pullback_candle

        if self._second_leg.direction == "buy":
            return candle.high > anchor.high

        if self._second_leg.direction == "sell":
            return candle.low < anchor.low

        return True

    def _reset(self) -> None:
        self.state = SP2LState.IDLE

        self._pre_breakout_candle = None
        self._pending_spike = None
        self._p_gap = None
        self._second_leg = None
        self._pullback_structure = None
        self._signal_bar = None

from dataclasses import dataclass
from datetime import datetime
from typing import Any


@dataclass(frozen=True)
class SP2LAuditEvent:
    event: str
    timestamp: datetime
    state: str
    setup_id: int | None = None
    direction: str | None = None
    result: str | None = None
    reason: str | None = None
    data: dict[str, Any] | None = None

from dataclasses import dataclass
from decimal import Decimal

from strategies.spike_event import SpikeEvent
from models.candle import Candle
from strategies.spike_measurements import SpikeMeasurements

@dataclass(frozen=True)
class SpikeDetectorConfig:
    """
    Working assumptions for Spike detection.

    These values are NOT claimed to be Poursamadi's exact parameters.
    They are baseline values that can later be tested and optimized.
    """

    lookback: int = 5

    minimum_range_ratio: Decimal = Decimal("1.8")
    minimum_body_ratio: Decimal = Decimal("1.5")
    minimum_body_ratio_of_range: Decimal = Decimal("0.65")
    maximum_closing_wick_ratio: Decimal = Decimal("0.20")



class SpikeDetector:
    def __init__(
        self,
        config: SpikeDetectorConfig | None = None,
    ):
        self.config = config or SpikeDetectorConfig()

        if self.config.lookback <= 0:
            raise ValueError("lookback must be greater than zero")

    def detect(
        self,
        candles: list[Candle] | tuple[Candle, ...],
    ) -> SpikeMeasurements | None:
        """
        Evaluate the latest completed candle against its recent context.

        The detector is timeframe-agnostic. The caller decides whether
        the supplied candles are M5, M15, or another timeframe.
        """

        required_candles = self.config.lookback + 1

        if len(candles) < required_candles:
            return None

        candidate = candles[-1]
        previous = candles[-required_candles:-1]

        average_range = self._average_range(previous)
        average_body = self._average_body(previous)

        return self._measure_candidate(
            candidate=candidate,
            average_range=average_range,
            average_body=average_body,
            previous=previous,
        )

    def _measure_candidate(
        self,
        candidate: Candle,
        average_range: Decimal,
        average_body: Decimal,
        previous: list[Candle] | tuple[Candle, ...],
    ) -> SpikeMeasurements:
        open_price = Decimal(str(candidate.open))
        high = Decimal(str(candidate.high))
        low = Decimal(str(candidate.low))
        close = Decimal(str(candidate.close))

        total_range = high - low
        body = abs(close - open_price)

        upper_wick = high - max(open_price, close)
        lower_wick = min(open_price, close) - low

        if total_range == 0:
            body_ratio_of_range = Decimal("0")
            closing_wick_ratio = Decimal("0")
        else:
            body_ratio_of_range = body / total_range

            if close >= open_price:
                closing_wick = upper_wick
            else:
                closing_wick = lower_wick

            closing_wick_ratio = closing_wick / total_range

        if average_range == 0:
            range_ratio = Decimal("0")
        else:
            range_ratio = total_range / average_range

        if average_body == 0:
            body_ratio = Decimal("0")
        else:
            body_ratio = body / average_body

        previous_high = max(
            Decimal(str(candle.high))
            for candle in previous
        )

        previous_low = min(
            Decimal(str(candle.low))
            for candle in previous
        )

        high_breakout_distance = high - previous_high
        low_breakout_distance = previous_low - low

        if close > open_price:
            direction = "buy"
            close_breakout_distance = close - previous_high
        elif close < open_price:
            direction = "sell"
            close_breakout_distance = previous_low - close
        else:
            direction = "none"
            close_breakout_distance = Decimal("0")
        breakout_distance = close_breakout_distance
        is_spike = (
            direction != "none"
            and range_ratio >= self.config.minimum_range_ratio
            and body_ratio >= self.config.minimum_body_ratio
            and body_ratio_of_range
            >= self.config.minimum_body_ratio_of_range
            and closing_wick_ratio
            <= self.config.maximum_closing_wick_ratio
            and close_breakout_distance > 0
        )

        return SpikeMeasurements(
            direction=direction,
            total_range=total_range,
            body=body,
            upper_wick=upper_wick,
            lower_wick=lower_wick,
            body_ratio_of_range=body_ratio_of_range,
            closing_wick_ratio=closing_wick_ratio,
            average_previous_range=average_range,
            average_previous_body=average_body,
            range_ratio=range_ratio,
            body_ratio=body_ratio,
            previous_high=previous_high,
            previous_low=previous_low,
            high_breakout_distance=high_breakout_distance,
            low_breakout_distance=low_breakout_distance,
            close_breakout_distance=close_breakout_distance,
            is_spike=is_spike,
            breakout_distance=breakout_distance,
        )

    @staticmethod
    def _average_range(
        candles: list[Candle] | tuple[Candle, ...],
    ) -> Decimal:
        ranges = [
            Decimal(str(candle.high)) - Decimal(str(candle.low))
            for candle in candles
        ]

        return sum(ranges, Decimal("0")) / Decimal(len(ranges))

    @staticmethod
    def _average_body(
        candles: list[Candle] | tuple[Candle, ...],
    ) -> Decimal:
        bodies = [
            abs(
                Decimal(str(candle.close))
                - Decimal(str(candle.open))
            )
            for candle in candles
        ]

        return sum(bodies, Decimal("0")) / Decimal(len(bodies))

    def detect_event(self, candles: list[Candle]) -> "SpikeEvent | None":

        measurements = self.detect(candles)

        if measurements is None or not measurements.is_spike:
            return None

        return SpikeEvent(
            breakout_candle=candles[-1],
            direction=measurements.direction,
            measurements=measurements,
        )

    previous: SwingPoint,
    current: SwingPoint,

    if (
        previous.type != "low"
        or current.type != "low"
    ):
        return None

            type="higher_low",
            candle=current.candle,
            price=current.price,

    return None

    previous: SwingPoint,
    current: SwingPoint,
) -> StructurePoint | None:
    if (
        previous.type != "high"

        return None

    if current.price < previous.price:
        return StructurePoint(
            type="lower_high",
            candle=current.candle,
            price=current.price,
        )

    return None

from strategies.structure import (
    StructurePoint,
    detect_higher_low,
    detect_lower_high,
)
from strategies.swing_history import SwingHistory


class StructureSequenceDetector:
    def __init__(self, history: SwingHistory):
        self.history = history

    def higher_low_sequence(
        self,
        minimum_length: int = 1,
    ) -> list[StructurePoint]:
        if minimum_length <= 0:
            raise ValueError(
                "minimum_length must be greater than zero"
            )

        lows = self.history.lows()

        if len(lows) < minimum_length + 1:
            return []

        sequence: list[StructurePoint] = []

        for previous, current in zip(lows[:-1], lows[1:]):
            structure = detect_higher_low(
                previous,
                current,
            )

            if structure is None:
                sequence = []
                continue

            sequence.append(structure)

        if len(sequence) < minimum_length:
            return []

        return sequence[-minimum_length:]

    def lower_high_sequence(
        self,
        minimum_length: int = 1,
    ) -> list[StructurePoint]:
        if minimum_length <= 0:
            raise ValueError(
                "minimum_length must be greater than zero"
            )

        highs = self.history.highs()

        if len(highs) < minimum_length + 1:
            return []

        sequence: list[StructurePoint] = []

        for previous, current in zip(highs[:-1], highs[1:]):
            structure = detect_lower_high(
                previous,
                current,
            )

            if structure is None:
                sequence = []
                continue

            sequence.append(structure)

        if len(sequence) < minimum_length:
            return []

        return sequence[-minimum_length:]

from dataclasses import dataclass
from decimal import Decimal

from models.candle import Candle


@dataclass(frozen=True)
class SwingPoint:
    type: str
    candle: Candle
    price: Decimal


class SwingDetector:
    def detect(self, candles: list[Candle]) -> SwingPoint | None:
        if len(candles) < 3:
            return None

        previous = candles[-3]
        candidate = candles[-2]
        current = candles[-1]
        if (
            candidate.high == previous.high
            or candidate.high == current.high
            or candidate.low == previous.low
            or candidate.low == current.low
            ):
                return None

        is_swing_high = (
            candidate.high > previous.high
            and candidate.high > current.high
        )

        is_swing_low = (
            candidate.low < previous.low
            and candidate.low < current.low
        )

        if is_swing_high:
            return SwingPoint(
                type="high",
                candle=candidate,
                price=Decimal(str(candidate.high)),
            )

        if is_swing_low:
            return SwingPoint(
                type="low",
                candle=candidate,
                price=Decimal(str(candidate.low)),
            )

        return None

class SwingHistory:
    def __init__(self):
        self._points: list[SwingPoint] = []

    def add(self, point: SwingPoint) -> None:
        self._points.append(point)

    @property
    def points(self) -> list[SwingPoint]:
        return list(self._points)

    def highs(self) -> list[SwingPoint]:
        return [
            point
            for point in self._points
            if point.type == "high"
        ]

    def lows(self) -> list[SwingPoint]:
        return [
            point
            for point in self._points
            if point.type == "low"
        ]

    def latest_high(self) -> SwingPoint | None:
        highs = self.highs()

        if not highs:
            return None

        return highs[-1]

    def latest_low(self) -> SwingPoint | None:
        lows = self.lows()

        if not lows:
            return None

        return lows[-1]

from collections import deque

from models.candle import Candle
from strategies.swing import SwingDetector, SwingPoint
from strategies.swing_history import SwingHistory


class SwingProcessor:
    def __init__(
        self,
        detector: SwingDetector | None = None,
        history: SwingHistory | None = None,
    ):
        self.detector = detector or SwingDetector()
        self.history = history or SwingHistory()

        self._candles: deque[Candle] = deque(maxlen=3)

    def process(self, candle: Candle) -> SwingPoint | None:
        self._candles.append(candle)

        if len(self._candles) < 3:
            return None

        point = self.detector.detect(list(self._candles))

        if point is None:
            return None

        self.history.add(point)
        return point
